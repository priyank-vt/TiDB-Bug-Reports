#!/usr/bin/env python3
"""Raw MySQL handshake client for testing the tidb_session_token auth plugin.
Usage: token_login.py <user> <plugin_name> <token_file> [--query "SQL"]
Connects to 127.0.0.1:4100, declares <plugin_name>, sends the token file content
as the auth response, prints OK/ERR, and optionally runs one COM_QUERY."""
import socket, struct, sys

def read_packet(s):
    hdr = b""
    while len(hdr) < 4:
        hdr += s.recv(4 - len(hdr))
    ln = hdr[0] | hdr[1] << 8 | hdr[2] << 16
    seq = hdr[3]
    payload = b""
    while len(payload) < ln:
        payload += s.recv(ln - len(payload))
    return seq, payload

def send_packet(s, seq, payload):
    s.sendall(struct.pack("<I", len(payload))[:3] + bytes([seq]) + payload)

def parse_greeting(p):
    i = 1
    while p[i] != 0: i += 1  # server version
    i += 1  # NUL
    i += 4  # thread id
    part1 = p[i:i+8]; i += 8
    i += 1  # filler
    caps = p[i] | p[i+1] << 8; i += 2  # capability lower
    i += 1 + 2 + 2  # charset, status, capability upper
    alen = p[i]; i += 1
    i += 10  # reserved
    n2 = max(13, alen - 8)
    part2 = p[i:i+n2]
    i += n2
    plugin = p[i:].rstrip(b"\x00").decode()
    return caps, plugin

def handshake(user, plugin, auth_data, query=None):
    s = socket.create_connection(("127.0.0.1", 4100), timeout=10)
    seq, p = read_packet(s)
    _, server_plugin = parse_greeting(p)
    caps = 0x1 | 0x200 | 0x2000 | 0x8000 | 0x80000 | 0x200000  # ...|PLUGIN_AUTH|PLUGIN_AUTH_LENENC_CLIENT_DATA
    resp = struct.pack("<IIB23x", caps, 1 << 24, 45)
    resp += user.encode() + b"\x00"
    if len(auth_data) < 251:
        resp += bytes([len(auth_data)]) + auth_data
    else:
        resp += b"\xFC" + struct.pack("<H", len(auth_data)) + auth_data
    resp += plugin.encode() + b"\x00"
    send_packet(s, seq + 1, resp)
    seq, p = read_packet(s)
    while p[0] == 0xFE and len(p) > 1:  # AuthSwitchRequest: replay same payload
        send_packet(s, seq + 1, auth_data)
        seq, p = read_packet(s)
    if p[0] == 0x00:
        print("OK")
        if query:
            s.settimeout(5)
            send_packet(s, 0, b"\x03" + query.encode())  # sequence resets to 0 per command
            rows = []
            ncols = None
            try:
                while True:
                    seq, p = read_packet(s)
                    if p[0] == 0xFF:
                        print("ERR:", p[3:].decode(errors="replace")); break
                    if p[0] == 0xFE and len(p) < 9:  # EOF
                        if ncols is None:
                            continue  # EOF after column defs
                        else:
                            break  # EOF after last row
                    if ncols is None:
                        ncols = p[0]; continue
                    # text row: lenenc strings; take first column value
                    i, vals = 0, []
                    while i < len(p):
                        n = p[i]; i += 1
                        vals.append(p[i:i+n].decode(errors="replace")); i += n
                    rows.append(vals)
            except socket.timeout:
                pass
            for r in rows:
                print("ROW:", r)
            if rows or ncols is not None:
                print(f"(result set: {len(rows)} rows, {ncols} columns)")
        s.close()
        return 0
    elif p[0] == 0xFF:
        code = p[1] | p[2] << 8
        print(f"ERR code={code}: {p[3:].decode(errors='replace')}")
        s.close()
        return 2
    print(f"UNEXPECTED packet: {p[:40].hex()}")
    s.close()
    return 3

if __name__ == "__main__":
    user, plugin, token_file = sys.argv[1], sys.argv[2], sys.argv[3]
    query = None
    if len(sys.argv) > 4 and sys.argv[4] == "--query":
        query = sys.argv[5]
    auth = open(token_file, "rb").read().strip()
    print(f"# user={user} plugin={plugin} auth_bytes={len(auth)}", file=sys.stderr)
    sys.exit(handshake(user, plugin, auth, query))
